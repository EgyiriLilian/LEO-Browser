Browser History Stack Class

class BrowserStack {
    constructor() {
        this.items = [];
    }

    push(element) {
        this.items.push(element); // O(1) Push to Top
    }

    pop() {
        if (this.isEmpty()) return null;
        return this.items.pop(); // O(1) LIFO Pop from Top
    }

    peek() {
        if (this.isEmpty()) return null;
        return this.items[this.items.length - 1]; // O(1) Top Inspection
    }

    isEmpty() {
        return this.items.length === 0;
    }

    size() {
        return this.items.length;
    }
}
